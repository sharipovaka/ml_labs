Признаки: distance_km и route_km — расстояние по прямой и длина маршрута, items и bill_rub — число позиций и сумма чека, queue — заказов в очереди у кухни, courier_load — загрузка курьера, %.

def make_delivery(n, seed):
    # время доставки = 8 + 3·км + 0.8·позиции + 0.25·очередь + 0.05·загрузка + шум
    rng = np.random.default_rng(seed)
    distance_km = np.clip(rng.gamma(2.0, 2.0, n), 0.3, 20)
    items = np.clip(np.round(rng.gamma(3.0, 1.5, n)), 1, 15)
    queue = rng.integers(0, 41, n).astype(float)
    courier_load = np.clip(rng.normal(70, 15, n), 20, 100)
    route_km = distance_km * 1.15 + rng.normal(0, 0.4, n)
    bill_rub = items * 450 + rng.normal(0, 250, n)
    minutes = 8 + 3.0 * distance_km + 0.8 * items + 0.25 * queue + 0.05 * courier_load + rng.normal(0, 3.0, n)
    return pd.DataFrame({'distance_km': distance_km.round(1), 'route_km': route_km.round(1), 'items': items,
                         'bill_rub': bill_rub.round(0), 'queue': queue, 'courier_load': courier_load.round(0),
                         'minutes': minutes.round(1)})

def gd_until(X, y, eta, eps, max_iter=100000):
    # спуск до остановки: шаг по θ стал короче eps
    theta = np.zeros(X.shape[1])
    for t in range(1, max_iter + 1):
        theta_new = theta - eta * 2 * X.T @ (X @ theta - y)
        if np.linalg.norm(theta_new - theta) <= eps:
            return theta_new, t
        theta = theta_new
    return theta, max_iter
